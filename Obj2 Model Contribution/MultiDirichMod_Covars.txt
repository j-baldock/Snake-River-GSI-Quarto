for (i in 1:Nmix) {
    Y[i,1:Nrgs] ~ dmulti(p[i,1:Nrgs], N[i])
    for (j in 1:Nrgs) {
        p[i,j] <- delta[i,j] / sum(delta[i,])
        delta[i,j] ~ dgamma(a[i,j], 1) T(0.0001,)
        log(a[i,j]) <- beta0 + beta1*dist[i,j] + beta2*area[j] + beta3*gw[j]
        loglik[i,j] <- logdensity.gamma(delta[i,j], a[i,j], 1) + logdensity.multi(Y[i,1:Nrgs], p[i,1:Nrgs], N[i])
    }
}
