for (i in 1:Nmix) {
    Y[i,1:Nrgs] ~ ddirch(a[i,1:Nrgs])
    loglik[i] <- logdensity.dirch(Y[i,1:Nrgs], a[i,1:Nrgs])
    for (j in 1:Nrgs) {
        log(a[i,j]) <- beta0 + beta1*dist[i,j] + beta2*area[j] + beta3*gw[j]
    }
}

beta3 ~ dnorm(0, pow(10, -2))
