model {

##-----------LIKELIHOOD-------------------##
for (i in 1:Nmix) {
    Y[i,1:Nrgs] ~ dmulti(p[i,1:Nrgs], N[i])
    for (j in 1:Nrgs) {
        logit(p[i,j]) <- beta0 + beta1*dist[i,j] + beta2*area[j] + beta3*gw[j]
    }
    loglik[i] <- logdensity.multi(Y[i,1:Nrgs], p[i,1:Nrgs], N[i])
}

##-----------PRIORS-----------------------##

# INTERCEPT
beta0 ~ dnorm(0, pow(10, -2))

# SLOPES
beta1 ~ dnorm(0, pow(10, -2))
beta2 ~ dnorm(0, pow(10, -2))
beta3 ~ dnorm(0, pow(10, -2))

##-----------DERIVED VALUES-----------------------##
for (i in 1:Nmix) {
    y_pred[i,1:Nrgs] ~ dmulti(p[i,1:Nrgs], N[i])
}

}
